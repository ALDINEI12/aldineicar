function calcularTotalCompraDinamica() {
    const inputQtd = document.getElementById('compraQtd');
    const txtTotal = document.getElementById('compraTotalExibicao');
    if (!inputQtd || !txtTotal) return;

    let qtd = parseInt(inputQtd.value) || 1;

    if (qtd > estoqueDisponivelSelecionado) {
        alert(`Quantidade máxima em estoque: ${estoqueDisponivelSelecionado}`);
        qtd = estoqueDisponivelSelecionado;
        inputQtd.value = qtd;
    }

    const total = qtd * precoUnitarioSelecionado;
    txtTotal.innerText = `R$ ${total.toFixed(2)}`;
}


function fecharModalCompraDinamica() {
    const modal = document.getElementById('modalCompraProdutoDinamico');
    if (modal) modal.remove();
}

async function processarBaixaEstoqueEPix() {
    const inputQtd = document.getElementById('compraQtd');
    const inputNome = document.getElementById('compraNomeCliente');
    const inputTel = document.getElementById('compraTelCliente');

    if (!inputNome || !inputTel || !inputNome.value.trim() || !inputTel.value.trim()) {
        alert("Por favor, preencha seu nome e telefone para contato!");
        return;
    }

    const qtdComprada = parseInt(inputQtd.value) || 1;
    const nomeCliente = inputNome.value.trim();
    const telCliente = inputTel.value.trim();
    const totalVenda = qtdComprada * precoUnitarioSelecionado;

    try {
        // Passo A: Busca a linha atualizada da sua oficina na tabela 'user_data'
        const { data: dadosAtuais, error: erroBusca } = await supabaseClient
            .from('user_data')
            .select('*')
            .eq('user_id', idOficinaDaLojaAtual)
            .single();

        if (erroBusca) throw new Error("Não foi possível conectar ao banco da oficina.");

        let listaMateriais = Array.isArray(dadosAtuais.materiais) ? dadosAtuais.materiais : [];
        let listaHistorico = Array.isArray(dadosAtuais.historico) ? dadosAtuais.historico : [];

        // Passo B: Localiza o produto nos seus materiais para subtrair do estoque
        let produtoEncontrado = false;
        listaMateriais = listaMateriais.map(item => {
            // Verifica o ID do material correspondente
            if (item.id === produtoSelecionadoParaCompra) {
                let estoqueAtual = parseInt(item.qtd || item.quantidade || 0);

                if (estoqueAtual < qtdComprada) {
                    throw new Error(`Estoque insuficiente. Disponível: ${estoqueAtual} un.`);
                }

                // Deduz o estoque nos formatos possíveis que seu sistema usa
                item.qtd = estoqueAtual - qtdComprada;
                if (item.hasOwnProperty('quantidade')) item.quantidade = estoqueAtual - qtdComprada;
                
                produtoEncontrado = true;
            }
            return item;
        });

        // Caso o produto venha de outra lista ou precise dar baixa direta na tabela 'produtos_loja'
        if (!produtoEncontrado) {
            // Tenta também deduzir direto se você usa a tabela secundária 'produtos_loja'
            const { data: prodLoja } = await supabaseClient
                .from('produtos_loja')
                .select('estoque')
                .eq('id', produtoSelecionadoParaCompra)
                .single();
                
            if (prodLoja) {
                const novoEstoqueLoja = (parseInt(prodLoja.estoque) || 0) - qtdComprada;
                await supabaseClient
                    .from('produtos_loja')
                    .update({ estoque: novoEstoqueLoja >= 0 ? novoEstoqueLoja : 0 })
                    .eq('id', produtoSelecionadoParaCompra);
            }
        }

        // Passo C: Injeta a venda no seu histórico no mesmo formato das OS
        const novoRegistroVenda = {
            data: new Date().toLocaleString('pt-BR'),
            status: 'Pago',
            cliente: { nome: nomeCliente, endereco: 'Compra Direta via Loja Virtual', tel: telCliente, cidade: 'Sátrio Dias/BA', city: 'Sátrio Dias/BA' },
            veiculo: { modelo: 'Venda de Balcão', placa: '---', ano: '', cor: '', avaliador: 'Vitrine Virtual', tipo_servico: `[COMPRA LOJA]: ${qtdComprada}x ${nomeProdutoSelecionado}` },
            materiais: [{ id: produtoSelecionadoParaCompra, nome: nomeProdutoSelecionado, qtd: qtdComprada, valor: precoUnitarioSelecionado }],
            mao_obra: [],
            totalMateriais: totalVenda,
            totalMateriaisCalculado: totalVenda,
            totalMaoObraCalculado: 0,
            deslocamento: 0,
            totalCobrado: totalVenda,
            lucro: totalVenda
        };
        listaHistorico.unshift(novoRegistroVenda);

        // Passo D: Atualiza a sua conta no 'user_data' salvando as alterações
        const { error: erroUpdate } = await supabaseClient
            .from('user_data')
            .update({
                materiais: listaMateriais,
                historico: listaHistorico,
                updated_at: new Date()
            })
            .eq('user_id', idOficinaDaLojaAtual);

        if (erroUpdate) throw erroUpdate;

        // Passo E: Redirecionamento e mensagem de sucesso para o Pix
        alert(`Pedido Confirmado, ${nomeCliente}!\nO estoque já foi atualizado no sistema da ALDINEICAR.\n\nEfetue o pagamento de R$ ${totalVenda.toFixed(2)} via PIX.\nChave Pix: INSIRA_AQUI_SEU_CELULAR_OU_CNPJ`);
        
        fecharModalCompraDinamica();

        // Recarrega a vitrine pública para atualizar o número de estoque na tela do cliente
        if (typeof carregarProdutosVitrinePublica === 'function') {
            carregarProdutosVitrinePublica(idOficinaDaLojaAtual);
        }

    } catch (err) {
        console.error("Erro na transação:", err);
        alert(`Erro ao processar compra: ${err.message}`);
    }
}


function fecharModalCompra() {
    const modal = document.getElementById('modalCompraProduto');
    if (modal) modal.style.display = 'none';
}

function calcularTotalCompra() {
    const inputQtd = document.getElementById('compraQtd');
    if (!inputQtd) return;

    let qtd = parseInt(inputQtd.value) || 1;

    if (qtd > estoqueDisponivelSelecionado) {
        alert(`Quantidade máxima em estoque: ${estoqueDisponivelSelecionado}`);
        qtd = estoqueDisponivelSelecionado;
        inputQtd.value = qtd;
    }

    const total = qtd * precoUnitarioSelecionado;
    if (document.getElementById('compraTotalExibicao')) {
        document.getElementById('compraTotalExibicao').innerText = `R$ ${total.toFixed(2)}`;
    }
}

// FUNÇÃO PRINCIPAL: DEBITA DO ESTOQUE E GERA A COBRANÇA
async function processarPagamentoPix() {
    const qtdComprada = parseInt(document.getElementById('compraQtd').value) || 1;
    const nomeCliente = document.getElementById('compraNomeCliente').value.trim();
    const telCliente = document.getElementById('compraTelCliente').value.trim();

    if (!nomeCliente || !telCliente) {
        alert("Por favor, preencha seu nome e telefone para contato!");
        return;
    }

    try {
        // 1. Busca os dados atualizados da oficina no banco (user_data)
        const { data: dadosAtuais, error: erroBusca } = await supabaseClient
            .from('user_data')
            .select('*')
            .eq('user_id', idOficinaDaLojaAtual)
            .single();

        if (erroBusca) throw new Error("Não foi possível conectar ao banco da oficina.");

        let listaMateriais = Array.isArray(dadosAtuais.materiais) ? dadosAtuais.materiais : [];
        let listaHistorico = Array.isArray(dadosAtuais.historico) ? dadosAtuais.historico : [];

        // 2. Encontra o produto na lista de materiais e atualiza a quantidade (estoque)
        let produtoAtualizado = false;
        listaMateriais = listaMateriais.map(item => {
            if (item.id === produtoSelecionadoParaCompra) {
                const estoqueAtual = parseInt(item.qtd || item.quantidade || 0);
                
                if (estoqueAtual < qtdComprada) {
                    throw new Error("O estoque deste produto mudou recentemente. Selecione uma quantidade menor.");
                }
                
                // Reduz o estoque baseado no que o cliente colocou
                if (item.hasOwnProperty('qtd')) item.qtd = estoqueAtual - qtdComprada;
                if (item.hasOwnProperty('quantidade')) item.quantidade = estoqueAtual - qtdComprada;
                
                produtoAtualizado = true;
            }
            return item;
        });

        if (!produtoAtualizado) throw new Error("Produto não localizado no inventário da oficina.");

        // 3. Cria um registro dessa venda no seu Histórico Administrativo para você saber quem comprou
        const totalVenda = qtdComprada * precoUnitarioSelecionado;
        const novoRegistroVenda = {
            data: new Date().toLocaleString('pt-BR'),
            status: 'Pago', // Como é Pix, entra como pago
            cliente: { nome: nomeCliente, endereco: 'Compra Direta via Loja Virtual', tel: telCliente, cidade: 'Sátrio Dias/BA', city: 'Sátrio Dias/BA' },
            veiculo: { modelo: 'Balcão / Loja', placa: '---', ano: '', cor: '', avaliador: 'Vitrine', tipo_servico: `[VENDA PRODUTO]: ${qtdComprada}x ${document.getElementById('compraProdutoNome').innerText}` },
            materiais: [{ id: produtoSelecionadoParaCompra, nome: document.getElementById('compraProdutoNome').innerText, qtd: qtdComprada, valor: precoUnitarioSelecionado }],
            mao_obra: [],
            totalMateriais: totalVenda,
            totalMateriaisCalculado: totalVenda,
            totalMaoObraCalculado: 0,
            deslocamento: 0,
            totalCobrado: totalVenda,
            lucro: totalVenda
        };
        listaHistorico.unshift(novoRegistroVenda);

        // 4. Salva tudo de volta na linha 'user_data' da sua oficina no Supabase
        const { error: erroUpdate } = await supabaseClient
            .from('user_data')
            .update({
                materiais: listaMateriais,
                historico: listaHistorico,
                updated_at: new Date()
            })
            .eq('user_id', idOficinaDaLojaAtual);

        if (erroUpdate) throw erroUpdate;

        // 5. DIRECIONA PARA A TELA DE PAGAMENTO PIX
        // Aqui você pode colocar o link da sua chave Pix ou copiar o código Copia e Cola.
        // Exemplo profissional: Mostra os dados do Pix e avisa o cliente.
        alert(`Pedido Confirmado, ${nomeCliente}!\nO estoque já foi atualizado no sistema da ALDINEICAR.\n\nPróximo passo: Efetue o pagamento de R$ ${totalVenda.toFixed(2)} via PIX.\nChave Pix: SEU_CELULAR_OU_CNPJ_AQUI`);
        
        fecharModalCompra();
        
        // Opcional: Atualiza a vitrine se o cliente continuar navegando para ver o estoque novo reduzido
        if (typeof carregarProdutosDaLoja === 'function') carregarProdutosDaLoja();

    } catch (err) {
        console.error("Erro ao processar venda:", err);
        alert(`Não foi possível finalizar a compra: ${err.message}`);
    }
}
